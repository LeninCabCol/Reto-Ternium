create view num_pt_hijo_cons as 

SELECT CONSIGNATARIO,ARMA_TARIMA,NUMERO_PARTE_HIJO,max(PIEZAS_POR_PAQ_HIJO) as PIEZAS_POR_PAQ_HIJO ,max(PESO_MAX_DESPACHO_HIJO) as PESO_MAX_DESPACHO_HIJO ,count(NUMERO_PARTE_HIJO) as Discos_pedidos, sum(PESO_NETO_DESP) as peso_neto_total  from ultimos13a7meses

where ARMA_TARIMA is not NUll and DESARMA_TARIMA is null and NUMERO_PARTE_HIJO is not NULL

group by ARMA_TARIMA, NUMERO_PARTE_HIJO, CONSIGNATARIO
